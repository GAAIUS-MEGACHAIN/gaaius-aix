```bash
# 1. Verify system
cd f:\gaaius-aiX\gaaius-ai
python run_production_test.py

  -d '{"project_id":"test","project_name":"Test","framework":"react","platforms":["web"]}'

# 4. Get status
curl http://127.0.0.1:8000/api/builds/JOB_ID/status
```
